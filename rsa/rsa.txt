
# ============================ Nguyên lý RSA ============================

- Người A muốn nhắn tin với người B nhưng tin nhắn phải được bảo mật => tin nhắn cần được mã hóa
- Người A sẽ đưa public key cho mọi người. Để người nào muốn giao tiếp với người A một cách an toàn.
- Người B sẽ lấy Public key để mã hóa đoạn tin nhắn, sau đó gửi lại cho người A. Lúc này người nào khác A sẽ không thể mở khóa được. Chỉ có A mới có thể mở khóa được vì A giữ Private key.


Nguyên lý RSA là một phương pháp mã hóa khóa công khai, được sử dụng rộng rãi trong bảo mật thông tin. Nó dựa trên việc sử dụng hai số nguyên tố lớn để tạo ra một khóa công khai và một khóa riêng tư. Dưới đây là các bước cơ bản của thuật toán RSA:

# 1. Chọn hai số nguyên tố lớn p và q.

# 2. Tính n = p * q, đây sẽ là mô-đun của khóa công khai và khóa riêng tư.

# 3. Tính phi(n) = (p-1)(q-1), đây là hàm Euler của n.

# 4. Chọn một số nguyên e sao cho 1 < e < phi(n) và e là số nguyên tố cùng nhau với phi(n). Số e này sẽ là khóa công khai.

# 5. Tính d sao cho d * e ≡ 1 (mod phi(n)). Số d này sẽ là khóa riêng tư.

# 6. Để mã hóa một thông điệp m, sử dụng công thức c = m^e mod n, trong đó c là bản mã.

# 7. Để giải mã bản mã c, sử dụng công thức m = c^d mod n, trong đó m là thông điệp gốc.


# ============================ Demo code ============================ 
# greatest common divisor
def gcd(a, b):
    """Compute the greatest common divisor of a and b using the Euclidean algorithm."""
    while b:
        a, b = b, a % b
    return a

# lowest common multiple
def lcm(a, b): 
    """Compute the least common multiple of a and b."""
    return abs(a * b) // gcd(a, b)

# modular inverse
# e x d = 1 (mod phi(n))

def modinv(e, phi):
    t, newt = 0, 1
    r, newr = phi, e
    while newr != 0:
        quotient = r // newr
        t, newt = newt, t - quotient * newt
        r, newr = newr, r - quotient * newr
    if r > 1:
        raise Exception('e dont have an inverse')
    if t < 0:
        t += phi
    return t

# is coprime -- Số nguyên tố cùng nhau hay ko
def is_coprime(a, b):
    return gcd(a, b) == 1

def rsa_keygen(p, q, e):
    n = p * q
    phi = (p-1)*(q-1)
    if not is_coprime(e, phi):
        raise Exception('e is coprime with phi(n)')
    d = modinv(e, phi)
    return (n, e, d)

def rsa_encrypt(m, e, n):
    return pow(m, e, n)

def rsa_decrypt(c, d, n):
    return pow(c, d, n)

def demo():
    # Step 1: Choose two prime numbers p and q
    p, q = 61, 53
    
    # Step 2: Compute n = p * q
    # Step 3: Compute phi(n) = (p-1)(q-1)
    # Step 4: Choose e such that 1 < e < phi(n) and e is coprime with phi(n)
    e = 17
    n, e, d = rsa_keygen(p, q, e)
    
    # Step 5: Compute d such that d * e ≡ 1 (mod phi(n))
    # Step 6: Encrypt a message m using c = m^e mod n
    m = 65
    c = rsa_encrypt(m, e, n)
    
    # Step 7: Decrypt the ciphertext c using m = c^d mod n
    m2 = rsa_decrypt(c, d, n)
    print(f"Plaintext: {m}, Ciphertext: {c}, Decrypted: {m2}")

if __name__ == "__main__":
    demo()
